# 多条件查询 must 相当于andGET kuangshen/user/_search{  "query": {    "bool": {      "must": [        {"match": {          "name": "狂神"        }},        {"match": {          "age": 23        }}      ]    }  }} # 多条件查询 should 相当于orGET kuangshen/user/_search{  "query": {    "bool": {      "should": [        {"match": {          "name": "狂神说"        }},        {"match": {          "age": 25        }}      ]    }  }} # 多条件查询 must_not 相当于 notGET kuangshen/user/_search{  "query": {    "bool": {      "must_not": [        {"match": {          "age": 25        }}      ]    }  }}  # 过滤查询1 age > 24GET kuangshen/user/_search{  "query": {    "bool": {      "must": [        {"match": {          "name": "狂神"        }}      ],      "filter": [        {"range": {          "age": {            "gt": 24          }        }}      ]    }  }} # 过滤器2  22<age<30 GET kuangshen/user/_search{  "query": {    "bool": {      "must": [        {"match": {          "name": "狂神"        }}      ],      "filter": [        {"range": {          "age": {            "lt": 30,            "gt": 22          }        }}      ]    }  }}
